7 Usage

Basic Range

<!-- Wrap in .mc-range-wrap with data-hue -->
<div class="mc-range-wrap" data-hue="blue">
  <input type="range" class="mc-range"
         min="0" max="100" value="50"
         style="--range-fill:50%;">
</div>

With Ticks & Labels

<div class="mc-range-wrap" data-hue="green">
  <input type="range" class="mc-range"
         min="0" max="100" step="10" value="60"
         style="--range-fill:60%;">
  <div class="mc-range-ticks">
    <span></span> <!-- repeat for each step -->
  </div>
  <div class="mc-range-labels">
    <span>0</span>
    <span>100</span>
  </div>
</div>

With Tooltip (JS)

// Update --range-fill and tooltip on input
input.addEventListener('input', () => {
  const pct = (input.value - input.min) / (input.max - input.min) * 100;
  input.style.setProperty('--range-fill', pct + '%');
  tooltip.textContent = input.value;
  tooltip.style.left = pct + '%';
});

Size Variants

<!-- Add size class to both wrapper and input -->
<div class="mc-range-wrap mc-range-wrap-lg" data-hue="red">
  <input type="range" class="mc-range mc-range-lg" ...>
</div>
