CSS Classes Reference 
/* Core Components */
.mc-panel          /* Container with header/body */
.mc-button         /* mc-button-[filled|outline|ghost] data-hue="[hue]" */
.mc-input          /* mc-input-[size] mc-input-[color] mc-input-mono */

/* Form Controls */
.mc-checkbox       /* Styled checkbox */
.mc-radio          /* Styled radio button */
.mc-switch         /* Toggle switch: mc-switch-[success|danger|warning] */
.mc-textarea       /* Multi-line input: mc-textarea-mono */
.mc-input-group    /* Input with mc-input-addon prefix/suffix */
.mc-select         /* Styled dropdown */

/* Feedback */
.mc-badge          /* mc-badge-[success|danger|warning|info] */
.mc-alert          /* mc-alert-[success|danger|warning|info] */
.mc-spinner        /* Loading: mc-spinner-[sm|lg|xl] mc-spinner-[color] */

/* Navigation */
.mc-tabs           /* Tab container with .mc-tab items */
.mc-breadcrumb     /* mc-breadcrumb-item, mc-breadcrumb-separator */
.mc-pagination     /* mc-page, .active, .disabled */

/* Data Display */
.mc-table          /* Styled table: mc-table-striped, mc-table-compact */
.mc-list-group     /* mc-list-item with .active, .disabled */
.mc-avatar         /* mc-avatar-[sm|lg|xl] */
.mc-progress       /* mc-progress-bar-[success|danger|warning|info] */

/* Utilities */
.mc-code           /* Code block: mc-code-inline for inline */
.mc-kbd            /* Keyboard key styling */
.mc-close          /* Close button: mc-close-[sm|lg] */
.mc-divider        /* Horizontal separator */

/* Validation */
.mc-input-error    /* Error state border */
.mc-input-success  /* Success state border */
.mc-error-text     /* Red helper text */
.mc-success-text   /* Green helper text */

/* Text Utilities */
.mc-text-positive  /* Green: gains, success */
.mc-text-negative  /* Red: losses, errors */
.mc-text-muted     /* Gray: secondary text */
.mc-text-label     /* Light gray: labels */
